def verification_status(checks):
    statuses = [c['status'] for c in checks]
    if 'failed' in statuses:
        return 'failed'
    if 'stale' in statuses:
        return 'stale'
    if statuses and all(s == 'passed' for s in statuses):
        return 'passed'
    if 'passed' in statuses:
        return 'partial'
    if 'interrupted' in statuses:
        return 'interrupted'
    if 'unverified' in statuses:
        return 'unverified'
    return 'not_run'


def snapshot(h, feature):
    plan, state = h.load(feature)
    try:
        criteria = h.criteria_digest(plan, state)
    except HarnessError:
        criteria = None
    source = h.source_digest(plan) if criteria else None
    checks = {}
    for check in plan['checks']:
        run = h.latest(state, 'green', check['id'])
        current = bool(run and criteria and h.evidence_current(run) and run['criteria_digest'] == criteria and run['source_digest'] == source)
        label = 'not_run' if not run else 'unverified' if run.get('pending') else 'stale' if not current else 'interrupted' if run.get('timed_out') else 'passed' if run['accepted'] else 'unverified' if run.get('pending') or run['exit_code'] is None else 'failed'
        log = ''
        output_truncated = False
        if run:
            path = Path(run['log']).resolve()
            if path.is_relative_to(h.folder(feature).resolve()) and path.is_file():
                with path.open(errors='replace') as stream:
                    log = stream.read(100001)
                output_truncated = len(log) > 100000
                log = log[:100000]
                if output_truncated:
                    log += '\n[출력 생략: 원본이 100,000자를 초과하여 앞부분만 포함했습니다.]'
            else:
                log = '[실행 출력 근거를 찾을 수 없습니다. 원본 로그를 복구하거나 다시 검사하세요.]'
        checks[check['id']] = {'id': check['id'], 'label': check.get('label', check['id']), 'kind': check['kind'],
                                'command': shlex.join(check['argv']), 'status': label, 'output': log, 'outputTruncated': output_truncated,
                                'exitCode': run.get('exit_code') if run else None, 'recordedAt': run.get('recorded_at') if run else None,
                                'files': check['files'], 'scope': check.get('scope', '등록된 검사 범위. 제품 화면 확인 여부는 검사 내용을 확인하세요.')}
    record = state.get('reviews', {}).get('implementation')
    review_current = bool(record and criteria and record.get('criteria_digest') == criteria and record.get('source_digest') == source)
    review_checks = {c['id']: c for c in (record or {}).get('answer', {}).get('checks', [])}
    rows = []
    for req in plan['requirements']:
        implementation = state.get('implementations', {}).get(req['id'], {'status': 'unknown', 'paths': [], 'remaining': '', 'note': ''})
        linked = [checks[c] for c in req['checks']]
        review = review_checks.get(req['id'])
        review_status = 'not_run' if not record else 'stale' if not review_current else 'passed' if review and review['status'] == 'pass' and record.get('passed') else 'failed' if (review and review['status'] == 'fail') or (record or {}).get('answer', {}).get('verdict') == 'fail' else 'unverified'

                     'limitations': plan['limitations'] + (record or {}).get('answer', {}).get('limitations', []) + [str(f) for f in (record or {}).get('answer', {}).get('findings', [])]})
    return rows


def record_implementation(h, feature, requirement, status, paths, remaining='', note=''):
    from .workflow import save_json
    plan, state = h.load(feature)
    require(requirement in [r['id'] for r in plan['requirements']], '알 수 없는 요구 ID')
    require(status in IMPLEMENTATION, '잘못된 구현 상태')
    require(all(isinstance(p, str) for p in paths), '구현 경로는 문자열 배열이어야 합니다')
    for path in paths:
        require(not Path(path).is_absolute(), '구현 경로는 프로젝트 상대 경로여야 합니다')
        require(project_path(h.root, path).exists(), f'구현 경로 없음: {path}')
    require(status != 'implemented' or paths, '구현 완료에는 구현 경로가 필요합니다')
    require(status != 'partial' or remaining.strip(), '일부 구현에는 남은 범위가 필요합니다')
    state.setdefault('implementations', {})[requirement] = {'status': status, 'paths': paths, 'remaining': remaining, 'note': note}
    # This is an implementation declaration, never a test/review success claim.
    save_json(h.folder(feature) / 'state.json', state)
    return state['implementations'][requirement]

export async function projectFile(rootPath: string, path: string): Promise<string> {
  if (typeof path !== 'string' || !path.trim() || isAbsolute(path)) throw new StoreError('Use a project-relative path');
  const root = await realpath(rootPath), candidate = await realpath(resolve(root, path)).catch(() => { throw new StoreError(`File not found: ${path}`, 404); });

async function changedAt(path: string, seen = new Set<string>()): Promise<number> {
  const actual = await realpath(path);
  if (seen.has(actual)) return 0;
  seen.add(actual);
  const info = await stat(path);
  if (!info.isDirectory()) return info.mtimeMs;
  let latest = info.mtimeMs;
  for (const child of await readdir(path, { withFileTypes: true })) {
    if (child.isSymbolicLink()) continue;
    if (['node_modules', '.git', '.proofstill', '__pycache__', '.pytest_cache', 'dist'].includes(child.name)) continue;
    latest = Math.max(latest, await changedAt(resolve(path, child.name), seen));
  }
  return latest;
}

    const result: HandoffSnapshot = { rows: [], errors: [], generatedAt: new Date().toISOString() };

    let hasHarness = false;
    try { hasHarness = (await stat(evidenceDir)).isDirectory(); } catch (error: any) { if (error.code !== 'ENOENT') result.errors.push({ taskId: 'harness', message: String(error) }); }
    if (hasHarness) {
      try { const data = await harness(project.rootPath, ['handoff']); result.rows.push(...data.rows); result.errors.push(...data.errors); }
      catch (error) { result.errors.push({ taskId: 'harness', message: `하네스 조회 실패. 해당 요구 상태를 확인하지 못했어요: ${error instanceof Error ? error.message : String(error)}` }); }

      }
    }

        return { id: check.id, label: check.label, kind: check.kind, command: check.command, status, output: e?.output ?? '', exitCode: e?.exitCode ?? null,
          recordedAt: e?.finishedAt, files: check.files ?? [], scope: check.label, nodeId: e ? graph.nodes.find(n => n.executionId === e.id)?.id : undefined };
      }));
      const relatedNodeIds = graph.edges.filter(e => e.sourceId === node.id || e.targetId === node.id).map(e => e.sourceId === node.id ? e.targetId : e.sourceId);
      const origins = graph.nodes.filter(n => relatedNodeIds.includes(n.id) && n.evidence).map(n => `${n.title}\n${n.evidence!.content}`);
      result.rows.push({ id: `console:${node.id}`, source: 'console', taskId: node.id, requirementId: node.id, nodeId: node.id, relatedNodeIds,
        title: node.title, group: node.kind === 'feature' ? '앱 기능' : '앱 요구사항', origin: origins.join('\n\n') || node.body,
        implementation, verification: { status: verificationStatus(checks.map(c => c.status)), checks, review: { status: 'not_run', evidence: '앱 실행 결과는 독립 리뷰가 아니에요.' } }, limitations });
    }
    return result;
  }
  async link(projectId: string, taskId: string, requirementId: string) {
    const row = (await this.snapshot(projectId)).rows.find(r => r.source === 'harness' && r.taskId === taskId && r.requirementId === requirementId);
    if (!row) throw new StoreError('하네스 요구를 찾을 수 없어요.', 404);
    return this.store.linkHarnessRequirement(projectId, taskId, requirementId, row.title);
  }
  async save(projectId: string, input: { source: string; taskId: string; requirementId: string; implementation: ImplementationProgress }) {
    if (!input || !['console', 'harness'].includes(input.source) || typeof input.taskId !== 'string' || typeof input.requirementId !== 'string') throw new StoreError('Invalid progress reference');
    const root = this.store.getProject(projectId).rootPath, p = input.implementation;
    if (!p || Object.keys(p).some(k => !['status', 'paths', 'remaining', 'note'].includes(k)) || !Object.hasOwn(implementationLabels, p.status) || !Array.isArray(p.paths) || p.paths.some(path => typeof path !== 'string') || typeof p.remaining !== 'string' || typeof p.note !== 'string') throw new StoreError('Invalid implementation progress');
    for (const path of p.paths) await projectFile(root, path);
    if (p.status === 'implemented' && !p.paths.length) throw new StoreError('구현 완료에는 코드 경로가 필요해요.');
    if (p.status === 'partial' && !p.remaining.trim()) throw new StoreError('일부 구현에는 남은 범위를 적어 주세요.');
    if (input.source === 'console') {
      if (input.taskId !== input.requirementId) throw new StoreError('Mismatched requirement reference');
      return this.store.saveImplementationProgress(projectId, input.taskId, p);
    }
    try { return await harness(root, ['implement', input.taskId, input.requirementId, '--status', p.status, ...p.paths.flatMap(path => ['--path', path]), '--remaining', p.remaining, '--note', p.note]); }
    catch (error) { throw new StoreError(error instanceof Error ? error.message : String(error), 409); }
  }
  async code(projectId: string, path: string) {
    const file = await projectFile(this.store.getProject(projectId).rootPath, path), info = await stat(file);
    if (info.isDirectory()) {
      const chunks: string[] = []; let size = 0, truncated = false;
      const walk = async (folder: string) => {
        for (const child of (await readdir(folder, { withFileTypes: true })).sort((a, b) => a.name.localeCompare(b.name))) {
          if (child.isSymbolicLink() || ['node_modules', '.git', '.proofstill', 'dist', '__pycache__', '.pytest_cache'].includes(child.name)) { chunks.push(`생략: ${child.name}`); continue; }
          const item = resolve(folder, child.name), rel = relative(this.store.getProject(projectId).rootPath, item);
          try {
            await projectFile(this.store.getProject(projectId).rootPath, rel);
            if (child.isDirectory()) { await walk(item); continue; }
            const itemSize = (await stat(item)).size;
            if (size + itemSize > 128 * 1024) { truncated = true; chunks.push(`크기 제한으로 생략: ${rel}`); continue; }
            size += itemSize; chunks.push(`--- ${rel} ---\n${await readFile(item, 'utf8')}`);
          } catch { chunks.push(`내용 확인 불가 또는 제외: ${rel}`); }
        }
      };
      await walk(file);
      return { path, content: chunks.join('\n\n') || '빈 폴더', truncated };
    }
    if (!info.isFile()) throw new StoreError('Not a regular file', 400);
    if (info.size > 128 * 1024) return { path, content: '128 KiB를 초과하여 내용을 포함하지 않았어요. 원본 파일을 확인해 주세요.', truncated: true };
    const content = await readFile(file, 'utf8');
    return { path, content, truncated: false };
  }

  buildContext(projectId: string, nodeIds: string[]): GraphContext {
    if (!Array.isArray(nodeIds) || !nodeIds.length || nodeIds.length > 100) throw new StoreError('Select 1 to 100 context nodes');
    const selected = new Set(nodeIds); for (const id of selected) this.getNode(projectId, id);
    const graph = this.listGraph(projectId), included = new Set(selected);
    const byId = new Map(graph.nodes.map(node => [node.id, node]));
    for (let depth = 0; depth < 2; depth++) {
      const frontier = new Set([...included].filter(id => selected.has(id) || byId.get(id)?.kind !== 'feature'));
      for (const edge of graph.edges) {
        const candidates = [frontier.has(edge.sourceId) ? edge.targetId : null, frontier.has(edge.targetId) ? edge.sourceId : null];
        for (const id of candidates) if (id && (depth === 0 || byId.get(id)?.kind !== 'feature' || selected.has(id))) included.add(id);
      }
    }
    const nodes = graph.nodes.filter(node => included.has(node.id));
    const edges = graph.edges.filter(edge => included.has(edge.sourceId) && included.has(edge.targetId));
    const context = nodes.map(node => node.evidence
      ? `EVIDENCE ${node.id} (${node.evidence.sourceKind})\nSOURCE ${json(node.evidence.source)}\nRAW\n${node.evidence.content}`
      : node.execution ? `EXECUTION ${node.id} (${node.execution.kind}, ${node.execution.status})\n${json(node.execution)}`
      : `NODE ${node.id} (${node.kind}, ${node.status}) ${node.title}\n${node.body}`).join('\n\n');
    const relations = edges.map(edge => `${edge.sourceId} ${edge.kind === 'precedes' ? '->' : '--'} ${edge.targetId}${edge.reason ? ': ' + edge.reason : ''}`).join('\n');
    return { projectId, nodeIds: [...selected], nodes, edges, text: `${context}\n\nRELATIONS\n${relations}` };
  }

          const write = (event: any) => { if (event.id > last) { last = event.id; response.write(`id: ${event.id}\ndata: ${JSON.stringify(event)}\n\n`); } };
          const listener = (item: any) => { if (item.sessionId === id) write(item.event); };
          service.on('event', listener); for (const event of replay) write(event);
          const heartbeat = setInterval(() => response.write(': heartbeat\n\n'), 15000); heartbeat.unref();
          response.on('close', () => { clearInterval(heartbeat); service.off('event', listener); streams.delete(response); }); return;
        }
      }
      throw new StoreError('Not found', 404);
    } catch (error) {
      if (response.headersSent) { response.end(); return; }
      const status = error instanceof StoreError ? error.status : error instanceof URIError || error instanceof SyntaxError ? 400 : String((error as any)?.code || '').startsWith('ERR_SQLITE') ? 409 : 502;
      json(response, status, { error: error instanceof Error ? error.message : String(error) });
    }
  });
  return { server, service, async close() { for (const stream of streams) stream.end(); service.close(); await new Promise<void>((resolve, reject) => server.close(error => error ? reject(error) : resolve())); } };
}

- 인계는 앱에서 근거 포함 HTML을 내려받는다. 원문·상태·남은 일·재현 명령·코드·출력이 들어가는 로컬 스냅샷이며, 누락/생략/조회 실패를 확인한다. 외부로 보내는 권한은 포함하지 않는다. 별도 장문 보고를 중복 작성하지 않는다.

## 빠른 구현과 사람이 이해할 수 있는 근거

- 가장 짧게 실행할 수 있는 `입력 → 검증 → 처리 → 저장 → 조회/내보내기` 사용자 흐름부터 연결하고 기능 단위로 확장한다. 실제 입력·계산·저장과 화면을 일찍 연결하며 미리보기 성공을 실제 동작으로 보고하지 않는다. 구현 중에는 변경 모듈 검사와 핵심 통합 검사를 우선하고 마지막에 필수 검사와 독립 리뷰를 모은다.
- 진행 설명은 `변경한 요구 ID / 핵심 조건과 처리 이유 / 실제 결과·검사 위치 / 남은 문제`로 짧게 전달한다. 중요한 분기에는 구체적인 입력과 기대 결과 하나를 들어 사람이 전체 로직을 따라갈 수 있게 한다. 이해 여부를 매번 퀴즈나 승인으로 확인하지 않는다.
- 복합 데이터 처리는 데이터 선택·검증·적용 → 계산 입력과 기준 → 결과·버전 저장 → 조회·내보내기의 코드 진입점을 연결한다. 날짜 경계·방향 비대칭·주소/좌표·참조 무결성·설정 변경 중 실제 요구에 해당하는 조건을 검사한다. 생성 알고리즘과 같은 계산을 반복해 정답으로 삼기보다 별도 불변조건 검사로 누락·중복·용량·시간 등 결과 규칙을 확인한다.
- 데이터 변경 검사는 정상 변경, 잘못된 입력, 경계 조건 중 이번 요구가 깨질 수 있는 사례를 고른다. 변경한 열·행·이유·예상 차이와 재현 명령을 연결하고 원본을 보존한다. 기대 결과를 고정한 가짜 처리나 모든 조합의 무차별 검사를 피한다. 알고리즘의 효율·편의 절충과 가정은 업무상 필수 규칙과 구분한다.
- 사용자가 AI 협업·구현 설명을 요구하면 실제 `지시/수정 → 달라진 동작 → 독립적으로 확인한 증거 → 발견한 문제와 조치`를 코드·명령·결과로 답할 수 있게 한다. 요구사항 기록과 필수 실행 안내/협업 제출물에 필요한 근거만 남기고 별도 장문 일지를 만들지 않는다. 실행하지 않은 검증·사용자의 판단·직접 극복한 경험을 꾸며 쓰지 않는다.

## 검사와 마무리

- 검증 단위는 파일 저장이나 매번의 코드 수정이 아니라 사용자가 확인할 수 있는 기능 묶음이다. 구현 중에는 불확실한 로직·수정한 결함에 필요한 빠른 검사만 실행한다. 매 변경마다 전체 테스트·빌드·브라우저·독립 리뷰를 순회하지 않는다. 새 변경·실패·미해결 우려가 없으면 통과한 검사를 반복하지 않는다.
- 문구·문서·스타일 등 동작 위험이 낮은 작은 변경은 직접 확인이나 관련 검사로 끝낸다. 하네스 작업 생성·기능 JSON·독립 리뷰·별도 완료 보고를 기본 의무로 만들지 않는다. 중요한 업무 규칙·권한·데이터 정합성·저장 경계가 바뀌거나 사용자가 독립 검증을 요청하면 기능 묶음의 최종 게이트를 적용한다. 변경한 범위와 확인한 범위를 짧게 알린다.
- 하네스를 사용하는 기능 묶음은 구현을 모은 뒤 현재 버전의 필수 검사와 독립 리뷰를 한 번 수행한다. 검사를 통과한 후 관련 코드를 다시 바꾸면 영향받는 검사를 확인한다. CLI는 작업 단위 전체 해시를 사용하므로 같은 작업의 최종 완료에는 등록한 모든 check를 현재 버전에서 다시 실행해야 한다. 이를 매 수정 직후 실행하라는 뜻으로 해석하지 않는다. sources를 임의로 빼거나 오래된 결과를 현재 통과로 취급하지 않는다. 무관한 기능은 처음부터 별도 작업으로 묶는다.

- 새 프로젝트나 중요한 기술 변경은 [시니어 개발자 인터뷰](../proofstill-engineer/SKILL.md)로 구현 가능성·주요 책임 경계를 사용자와 논의한다. 필요한 사용자 결정을 받은 뒤 스택·폴더 구조를 확정하고, 그에 맞는 스캐폴드·설치·폴더 생성과 구현으로 이어간다. 기존 합의 안의 작은 수정은 재인터뷰하지 않는다. 실제 답변·위임 범위·남은 위험을 기능 정의의 `context`에 보존한다. 미정 결정에 의존하는 생성·구현은 보류한다.
- 하네스 최종 게이트를 적용하는 기능은 요청과 관련 자료를 읽어 예제 `../../../examples/orders/feature.json`처럼 기능 정의를 만든다. `intent`에는 사용자 원문과 요구의 의미를 보존한다. 코드를 보고 사용자 의도를 추측하지 않는다.
- 요구사항별 ID와 check ID를 연결한다. 각 check에 정확한 argv, unit/integration 구분, module, 테스트/fixture 파일, 실제 실행을 확인할 Green 패턴을 지정한다. Red를 실행할 때만 의도한 실패 패턴을 지정한다. 의미 있는 부정 사례도 테스트한다. 외부 도구·공유 자원은 unit에서 격리하고 integration은 연결 경계에 둔다.
- 여러 단계의 출처 연결 자체를 검사해야 하거나 기존 Trace 계약이 있는 기능은 `Flow.step`으로 실제 함수를 계측하고 이전 출력은 `step.ref`, 컨텍스트는 `flow.context_ref`로 전달한다. 단계마다 작업 설명을 남긴다. 기대 순서·반복·데이터 값·출처 연결을 JSON으로 먼저 정하고 check의 `trace_expectation`으로 지정한다. 같은 값의 literal 복사는 출처 검증을 대체하지 않는다.
- 화면이 있는 기능은 이번에 바뀌는 사용자 작업의 IA와 상태 소유·수명만 함께 정한다. 언제 무엇을 조회하고, 무엇을 유지·갱신·제거할지, 초안 이탈·실패·재진입 정책을 임시 브리핑에서 먼저 확정한다. 파생값·다시 조회할 데이터의 중복 저장을 줄인다. 영구 설계 문서를 추가하지 않는다.
- IA·상태는 작은 화면·상태 흐름도 하나와 `유지 / 임시 / 정리` 몇 줄로 표현한다. 상태 이름 옆에 소유 범위와 종료 조건만 붙인다. 결정이 더 필요할 때만 확장한다. 테스트·실행 기준에 있는 내용을 표·장문·별도 문서로 중복하지 않고 파일 위치로 연결한다. JSON 기준은 항목별 한 줄 등 간결한 형식을 사용한다. 구현·검증 중에는 관련 부분만 읽고 대화에는 요약·변경분·근거 링크만 전달한다.
- 복잡한 상태 수명·복구·경계 누수가 핵심 위험이거나 기존 상태머신 계약이 있으면 UX 흐름을 상태머신 계약과 실제 Trace로 연결한다. 단순 화면은 직접 모듈·통합/UI 검사로 충분하면 별도 Trace 계약을 만들지 않는다. Trace를 사용하는 경우 다음을 따른다. 앱이 소유한 상태를 읽는 snapshot(`state_id/feature/screen/data`)과 `Flow.transition(..., transition_id=UX전이ID, state=앱.snapshot)`을 사용한다. 기대 JSON의 `state.initial`, `state.states`, ID별 `state.transitions`로 초기 상태·허용 전이·동작·기능/화면·필드와 `preserve/drop`을 검사한다. 중요한 전이는 check별 `required_transitions`에 넣어 실행 누락을 거부하고, `unvisited_transitions`를 통과 증거로 쓰지 않는다. 실행 예시는 `../../../examples/ux_save/expected.json`과 같은 폴더의 `demo.py`다. 기존 `state.screens`와 단계 번호 기반 전이 계약도 지원하지만 UX 상태머신 연동을 대체하지 않는다.
- 상태는 앱이 소유한다. snapshot에서 기대 ID를 임의로 붙이거나 Trace가 상태를 대신 변경하지 않는다. 조회 인자·저장 후 재조회, 초안 혼입·기능 종료 잔존·실패 시 손실·늦은 응답은 관련 모듈의 검사로 검증한다. 기존 수명 예시는 `../../../examples/order_workspace/expected.json`과 같은 폴더의 `workspace.py`, `demo.py`다. 관측 함수가 노출한 데이터만 기록·검증하므로 필요한 최소 JSON만 반환한다.
- CLI는 실행별 `PROOFSTILL_TRACE_DIR`에 생성된 JSONL을 수집·검증하고 HTML 조회기와 보고에 연결한다. 필요한 JSON 데이터만 명시적으로 기록한다. UI의 screen/action/element 매핑은 선언이며 실제 제품 UI 클릭 검증과 구분한다. 흐름 뷰어 생성은 `trace view <JSONL> --expect <기준 JSON> --output <HTML>`이다.
- `sources`에는 해당 기능에 영향을 줄 수 있는 구현·설정 경로를 모두 넣는다. 디렉터리와 아직 없는 파일도 지정할 수 있다. `context_files`에는 필요한 원래 요구·업무 맥락·담당자 실제 답변 근거를 넣는다. 요구사항 문서의 상태 표시는 성공 주장으로 믿지 않고 연결된 원문·답변·코드·검사를 대조한다. 전체 개발 대화·성공 주장·비밀은 리뷰 입력에 넣지 않는다. 제출 필수물과 기타 결과는 `artifacts`로 연결한다.
- `init <기능 정의 JSON>` 뒤 구현과 필요한 검사를 진행한다. `test <ID>`는 Green 검사를 실행하며 `--kind`와 `--module`로 범위를 좁힌다. 사전 기준 리뷰와 Red는 기본 필수 단계가 아니다. 의도한 실패를 먼저 확인할 가치가 있을 때만 `test <ID> red`를 사용한다.
- HTTP·CLI 연결이 바뀌거나 해당 경계의 확인이 필요할 때 `npm run test:integration` 또는 `npm run test:flow` 중 관련 검사를 메모리 DB+seed에서 실행한다. 두 명령을 매 변경마다 실행하지 않는다. `test:fast`는 unit/integration/flow를 실행한다. 각 실행은 임의 포트·전용 메모리 DB·임시 기록을 사용하며 개발 DB와 공유 flow 기록을 사용하지 않는다.
- 하네스를 사용하는 기능 묶음의 종료 전 모든 필수 check를 현재 소스·테스트에서 Green으로 확인하고 `review <ID>`로 기준 적절성과 구현을 새 Codex 세션에서 한 번에 검토한다. 별도 요청이나 중요한 미정 기준이 있을 때만 선택적으로 `review <ID> criteria`를 쓴다. 테스트 파일을 고치면 기존 검사·리뷰는 현재 완료 근거로 쓸 수 없으며, 추가 구현을 모은 뒤 같은 작업에서 최종 검사·리뷰한다. 요구사항이나 명령을 바꾸면 새 작업 ID로 정의한다.
- 리뷰는 원본 작업 폴더에서 실행하며 파일을 복사하지 않는다. 검토 대상 소스·기준·테스트는 편집하지 않고 임시 쓰기는 실행별 TMPDIR에 둔다. 리뷰 중 해당 작업의 대상이 바뀌면 그 판정을 완료 근거로 쓰지 않고 수정을 모아 최종 검사·리뷰한다. 독립성은 별도 세션과 직접 검사에서 확보하며 공유 파일의 보안 격리를 의미하지 않는다.
- 병렬 작업은 앱·기능별 파일 소유자를 나누고 각 작업 ID의 `sources`에 담당 구현과 실제 공유 의존성을 포함한다. 무관한 앱 전체를 넣지 않는다. 계약·공유 UI·migration·lockfile·생성 코드와 진입점 연결은 한 담당자가 순서대로 반영한다. 같은 작업 ID 명령은 직렬 실행한다. 폴더 분리와 실행별 자원으로 해결되지 않을 때만 Git worktree를 마지막 수단으로 고려하며 자동 생성하지 않는다.
- `report <ID>`가 기능별 Markdown 보고를 만든다. 코드·테스트·실행 명령·Red/Green 로그·독립 판정·한계를 한곳에서 찾아갈 수 있다. 보고를 사용자에게 전달한 뒤 `cleanup <ID>`로 임시 상태만 제거한다. 보고·증거·회귀 테스트는 남는다. 사용자 수용은 별개다.
- 인터뷰에서 확정한 요구와 실제 답변 근거를 기능 정의에 보존하고 검토 입력에 포함한다. 개발 완료·결과 전달 후 연결된 `.proofstill/work/intake-<ID>/brief.md`도 인터뷰 소유 파일임을 확인해 제거한다. 다른 파일이 있으면 보존하며 폴더 전체를 재귀 삭제하지 않는다.
- 현재 임시 상태는 `.proofstill/work/<ID>/brief.md`에서 재개하고 실제 코드와 대조한다. 하네스는 재개·추가 검사·보고 재생성을 지원한다. 한 작업 ID의 명령은 직렬 실행한다.
- 공유 로컬 파일을 악의적으로 조작하는 개발자를 격리하는 보안 경계는 아니다. CLI는 테스트를 직접 실행하므로 명령은 허용된 프로젝트 범위인지 먼저 확인한다. 명령 실패·누락·환경 오류는 통과로 취급하지 않는다. 모든 버그가 없다고 주장하지 말고 검증 범위와 남은 한계를 보고한다.

Node 22.18+와 Python 3.10+가 필요합니다. GitHub 명령에는 Git, 독립 리뷰에는 로그인한 Codex CLI가 필요합니다.

```sh
npx github:jwsong98/proofstill
```

긴 과제·명세는 원문을 보존하고 `requirements.md` 하나에 요구 ID, 조건, 원문·담당자 답변 근거, 구현·검사 위치와 상태를 모읍니다. 명확한 요구는 바로 구현하며 데이터에서 발견한 미정 업무 조건만 질문합니다. 추가 답변은 영향받는 부분부터 갱신하고, 입력→검증→처리→저장→조회 흐름을 일찍 연결합니다. 진행 설명과 필수 협업 제출물은 실제 변경 이유·검사 증거·한계에 연결합니다. 별도 Trace 계측은 복잡한 상태·출처 연결을 확인할 때 적용합니다. 이번 변경의 구현 시간 단축 효과는 아직 측정하지 않았습니다.

## 로컬 웹 콘솔

개발 체크아웃에서 Node 24+와 로그인한 Codex CLI로 실행합니다. 콘솔은 기존 전역 설치본을 바꾸지 않습니다.

```sh
npm run console:setup
npm run console
```

[로컬 콘솔](http://127.0.0.1:7777)에서 프로젝트 폴더를 등록하고, 문서 원문이나 선택한 Codex 세션을 가져옵니다. 원문 근거를 바탕으로 AI가 그래프를 연결하며, 선택한 맥락의 일반 Codex 채팅으로 이어갈 수 있습니다. 별도 작업 시작·자동 후속 실행 기능은 없습니다. 근거 원문은 읽기 전용으로 보존하고 해석·관계와 분리합니다.

기본 데이터는 `~/.local/share/proofstill/console/console.sqlite`에 남습니다. `PROOFSTILL_CONSOLE_DATA_DIR`로 보관 폴더, `PROOFSTILL_CONSOLE_PORT`로 포트를 바꿀 수 있습니다. `PROOFSTILL_CONSOLE_PROJECT_ROOT`를 지정하면 해당 프로젝트를 엽니다. Codex 인증은 로컬 app-server가 사용하며 브라우저에 인증 파일이나 API 키를 전달하지 않습니다.

프로젝트를 열면 **구현·검증 현황**에서 요구별 구현 상태, 자동 검사, 독립 리뷰, 남은 일을 따로 봅니다. 요구를 선택해 원문·구현 파일·실행 명령과 출력을 확인하고 구현 기록을 수정할 수 있습니다. 기존 항목은 구현 상태를 추정하지 않고 미기록으로 표시합니다. 앱의 기록·검사 연결은 SQLite에, 하네스 작업의 기록은 프로젝트 `.proofstill/evidence`에 보존합니다.

하네스 작업은 CLI와 앱이 같은 원본을 읽습니다. 예를 들어 `python3 -m proofstill --root /path/to/project implement task-id R1 --status partial --path src/upload.py --remaining "잘못된 파일 처리"`로 기록하고 `python3 -m proofstill --root /path/to/project handoff task-id`로 조회합니다. 구현 선언과 검사/리뷰 통과는 별개입니다. 기존 하네스 소스·기준 버전 검사를 유지하며, 앱 자체 요구의 검사는 요구/구현 기록과 지정 코드·검사 파일의 수정 시각으로 현재성을 확인합니다. 시각이 보존된 외부 변경은 감지하지 못합니다. 앱의 ‘근거 포함 인계 파일 받기’는 현재 상태·원문·구현 파일·재현 명령·출력을 담은 로컬 HTML을 생성합니다. 큰 파일·폴더·출력의 생략과 조회 오류는 인계 파일에서 확인해야 합니다. 다운로드한 시점 이후의 변경은 반영하지 않습니다. 구현 속도 개선 효과는 아직 측정하지 않았습니다.

콘솔 검사는 `npm run console:test`, `npm --prefix console run typecheck`, `npm run console:build`입니다. 메모리·임시 SQLite와 임의 포트에서 DB/API를 검사하며, 실제 Codex 모델 연결 검사는 테스트용 프로토콜 검사와 구분합니다. 문서 가져오기는 프로젝트 안의 UTF-8 텍스트 파일(최대 2 MiB)을 지원합니다.

직접 수정하려면 다음과 같이 사용합니다.

```sh
git clone https://github.com/jwsong98/proofstill.git
cd proofstill
npm install
node bin/proofstill.mjs setup
```

수정 후 setup을 다시 실행하면 전역 설치본이 갱신됩니다. 설치본은 `CODEX_HOME/proofstill`(기본 `~/.codex/proofstill`)에 보관하며, 스킬은 `~/.agents/skills`에 연결합니다. `setup --codex-home <경로> --skills-dir <경로>`로 위치를 지정할 수 있습니다. 스킬은 자동 감지되며 전역 AGENTS 지침은 새 세션에서 읽습니다. `--help`는 설정을 바꾸지 않습니다.

저장소 검사는 `npm test`, `npm run test:setup`, `npm run test:bootstrap`으로 실행합니다. bootstrap 검사는 먼저 `mkdir -p dist && npm pack --pack-destination dist`로 현재 버전의 tgz를 만든 뒤 실행합니다. Docker가 있는 환경에서는 `npm run test:database`로 Testcontainers까지 확인합니다.

npm으로 설치하는 개발·검증 하네스와 모노레포 스캐폴드입니다. 개인 API 키나 기존 사용자 스킬 폴더 없이 실행합니다. `doctor`는 실행 파일과 버전을 점검하며 Codex 로그인 상태를 보장하지는 않습니다.

```sh
npm install -g https://github.com/jwsong98/proofstill/releases/download/v0.5.0/digitalpresso-proofstill-0.5.0.tgz
proofstill doctor
proofstill scaffold my-project --package-manager npm --install
cd my-project
npm run build
npm run typecheck
npm run dev:backend
# 다른 터미널: npm run dev:app
```

에이전트에게 프로젝트 생성을 요청하면 기획·UX 확인 후 시니어 개발자 인터뷰로 구현 가능성과 아키텍처를 논의합니다. 필요한 사용자 결정을 받은 뒤 개발 스택과 폴더 구조를 확정하며, 제공 템플릿과 맞을 때 위 명령을 실행합니다. CLI 직접 실행은 인터뷰를 자동 수행하거나 차단하지 않습니다.

pnpm이 설치된 환경에서는 `--package-manager pnpm --install`을 사용합니다. 전역 설치 대신 `npm exec --yes --package=https://github.com/jwsong98/proofstill/releases/download/v0.5.0/digitalpresso-proofstill-0.5.0.tgz -- proofstill scaffold my-project --install`도 사용할 수 있습니다. GitHub 공개 Release에서 로그인 없이 내려받습니다. npm registry에는 아직 게시하지 않았습니다. 다운로드한 tgz는 `npm install -g ./digitalpresso-proofstill-0.5.0.tgz`로도 설치할 수 있습니다.

`--apps backend,cli,app`로 생성할 앱을 선택합니다. 기존 디렉터리(빈 디렉터리 포함)를 덮어쓰지 않습니다. 기본 백엔드는 FastAPI + Uvicorn + SQLAlchemy 2 ORM/Alembic, 웹 앱은 React/Vite입니다.

- `apps/backend/domain`: 업무 함수. `infra/app.py`: FastAPI 라우트와 Session 의존성. `infra/models.py`: ORM 모델. `infra/schemas.py`: 요청·응답 타입. 서버 실행 후 `/docs`·`/redoc`를 제공합니다.
- `packages/jwdesign`: 실제 컴포넌트·타입·폰트·토큰. 번들러용 React/portal import를 적용합니다.
- `packages/ui`: Radix Slot/CVA 기반 공유 버튼, jwdesign 토큰을 사용하는 shadcn 설정. `npm run ui:add -- card`로 확장합니다.
- `packages/api`: 수용한 계약, 서버 생성 명세, Orval TypeScript 클라이언트.
- `.agents/skills`: jwdesign, design-taste-frontend, proofstill-develop, proofstill-verify, proofstill-interview, proofstill-engineer, proofstill-ux, proofstill-api, proofstill-db.

생성 프로젝트는 하네스 배포본을 `.proofstill-runtime`에 포함해 공개 registry나 개인 경로 없이 다시 설치할 수 있습니다. npm/pnpm lockfile은 첫 설치에서 생성되므로 저장소에 함께 커밋하세요.

```sh
npm run openapi:export
npm run api:generate
npm run api:tracker -- env prepare --generate-spec
npm run api:tracker -- spec compare
npm run api:tracker -- flow validate
npm run api:tracker -- flow show
npm run api:tracker -- flow run
npm run api:tracker -- flow show
npm run api:tracker -- env stop
```

flow는 하나이며 별도 session 이름이나 ID를 선택하지 않습니다. 예제는 **할 일 등록 → 완료 처리 → 저장 결과 확인**입니다. `.api-tracker/flow.yaml`의 각 step에 사용자 행동(title), 목적(description), 기대 결과(expected), OpenAPI operationId를 작성합니다. `flow show`는 실행 전 안내와 실행 후 요청·응답·연결 출처를 함께 보여줍니다.

- `flow run`: 아직 실행하지 않은 단계를 순서대로 실행합니다. 완료 단계는 다시 호출하지 않습니다.
- `flow step create_task`: 다음 단계 하나만 실행합니다. 다른 단계로 건너뛸 수 없습니다.
- `flow retry`: 실패하거나 중단된 다음 단계를 명시적으로 재시도합니다. 서버에 이미 쓰기가 반영됐는지 먼저 확인하세요. 이전 시도는 같은 단계에 보존됩니다.
- `flow reset`: `.api-tracker/flow-record.json`만 제거합니다. 서버 데이터를 되돌리지 않습니다. 정의·명세·서버가 변경됐으면 기존 기록을 확인한 뒤 초기화합니다.

앞 단계의 실제 응답은 `{"from":"create_task","pointer":"/id"}`로 후속 step의 input.path/query/body에 연결합니다. 기대 응답은 expect.status와 expect.body의 JSON Pointer별 값으로 검사합니다. 실패하면 후속 단계는 실행하지 않습니다. 동시에 실행하면 flow.lock으로 중복 호출을 막습니다. 강제 종료로 잠금이 남으면 실행 프로세스 종료와 서버 처리 결과를 확인한 뒤 잠금 파일을 제거하고 명시적으로 재시도합니다.

기본 할 일 데이터는 서버 메모리에 있으며 재시작 시 사라집니다. 다른 서버는 flow의 baseUrl 또는 API_TRACKER_BASE_URL로 지정하고 spec 경로를 설정합니다. 인증은 headers의 `{"Authorization":{"env":"APP_AUTHORIZATION"}}`처럼 환경변수로 연결하며 헤더 값은 기록하지 않습니다. 이 버전은 순차 HTTP JSON 흐름을 지원하며 조건 분기·반복·병렬 실행은 포함하지 않습니다.

`proofstill.config.json`의 `openapi`에는 `command` argv, `file`, `url` 중 하나를 지정합니다. JSON/YAML 입력을 검증하여 JSON으로 내보내며 exporter 명령은 stdout에 명세만 출력해야 합니다. 각 API operation에는 고유한 `operationId`가 필요합니다. 외부 `$ref`는 먼저 하나의 문서로 묶어야 합니다. export는 계약을 자동으로 덮어쓰지 않습니다.

기존 저장소에서는 tgz를 devDependency로 설치한 후 `npx proofstill openapi export --url http://127.0.0.1:8000/openapi.json --output packages/api/openapi.generated.json` 또는 `--file spec.yaml`로 사용합니다. 명시적으로 계약을 가져올 때만 `--output packages/api/openapi.contract.json`을 지정합니다. 기존 저장소의 flow는 생성 예제의 `.api-tracker/flow.yaml`을 가져와 spec·baseUrl·steps를 해당 API에 맞게 연결합니다. 다른 백엔드를 연결할 때는 exporter와 lifecycle adapter를 해당 서버에 맞게 구현해야 합니다.

기존 Python 하네스는 `proofstill harness --help` 또는 생성 저장소의 `npm run proofstill -- --help`로 사용합니다. Python 실행 파일을 바꾸려면 `PROOFSTILL_PYTHON`을 지정합니다. 백엔드와 OpenAPI exporter는 `backend:setup`으로 만든 전용 Python을 함께 사용합니다. 실행 파일을 바꾸려면 `PROOFSTILL_BACKEND_PYTHON`을 지정합니다. Codex 설치·로그인과 Python/Node 자체 설치는 자동 수행하지 않습니다.

서버 lifecycle과 명세 비교는 기존 API Tracker 0.2.0의 배포 파일을 사용합니다(원본 커밋 `d7453f896263e1ed48f8e425613fed00edb4dec0`). jwdesign과 design-taste-frontend는 함께 제공된 원본 스킬·자산을 포함하며 폰트의 OFL 고지를 보존합니다. flow 호출·기록은 이 패키지의 단순한 순차 실행기를 사용합니다. 전체 API 스키마 적합성이나 실제 제품 UI 클릭까지 보증하지 않습니다.

현재 검증 대상은 macOS/POSIX, Node 24와 Python 3.14 환경의 npm/pnpm 설치·빌드·타입 검사·로컬 HTTP 흐름입니다. Windows와 제품 UI 클릭은 검증 범위 밖입니다. 시험 문제·파일·결과물을 외부로 전송하지 않으며 로컬 runtime/flow 기록은 소스 관리에서 제외합니다.

설정 형식의 근거: [shadcn monorepo](https://ui.shadcn.com/docs/monorepo), [Orval configuration](https://orval.dev/docs/reference/configuration/), [npm package files](https://docs.npmjs.com/cli/configuring-npm/package-json/).

## DB 명령과 검증 단계

`npm run backend:setup`으로 Python 의존성을 전용 `.venv`에 설치합니다. `scaffold --install`은 이 과정도 실행합니다. 아래 명령은 npm/pnpm 모두 사용할 수 있습니다.

| 명령 | 동작 |
|---|---|
| `backend:setup` | Python 가상환경과 고정 의존성 설치 |
| `db:check` | SQLite `:memory:`에서 실제 migration·seed 검사 |
| `test:unit` | 메모리 DB의 도메인·SQL 제약·migration·seed 검사 |
| `test:integration` | seed를 적용한 메모리 DB의 실제 HTTP 검사 |
| `test:flow` | CLI 사용자 흐름을 전용 메모리 DB·임의 포트·임시 기록으로 검사 |
| `test:fast` | unit + integration + flow, Docker 불필요 |
| `test:e2e` | PostgreSQL Testcontainers + 실제 HTTP flow + DB 값·지속성·정리 검사 |
| `db:revision -- --message "change"` | migration 생성, `--autogenerate` 지원 |
| `db:migrate` | migration 적용, `--revision`으로 목표 지정 |
| `db:status` | 현재·최신·미적용 revision 확인 |
| `db:rollback` | 한 단계 되돌리기, `--revision` 지정 가능 |
| `db:seed` | 고정 seed의 누락 ID 추가, 기존 수정값 보존 |
| `db:reset -- --yes` | 대상 DB의 전체 downgrade → upgrade → seed |

관리용 db:revision/migrate/status/rollback/seed/reset은 `DATABASE_URL`을 명시해야 합니다. 예를 들어 `export DATABASE_URL='sqlite+pysqlite:///./dev.sqlite'`로 별도 개발 DB를 지정합니다(상대 경로는 apps/backend 기준). PostgreSQL은 `postgresql+psycopg://…` URL을 사용합니다. `db:reset -- --yes`는 대상 DB 데이터를 제거합니다.

기본 서버는 매번 SQLite :memory:에 같은 Alembic migration과 `seed/tasks.json`을 적용합니다. 외부 DATABASE_URL을 지정한 서버는 최신 스키마인지 확인하며 자동으로 migration/seed를 수행하지 않습니다. 먼저 db:migrate와 db:seed를 실행합니다. test:unit/integration/flow/fast와 db:check는 외부 DB 설정을 무시합니다.

test:e2e는 메모리 DB의 unit/integration/flow를 먼저 실행하고 통과한 경우에만 컨테이너 단계로 진행합니다. E2E는 실행 중인 Docker가 필요합니다. Testcontainers가 임시 PostgreSQL을 만들고 migration·seed, 실제 API 사용자 흐름, SQL 행 일치와 서버 재시작 지속성까지 확인한 뒤 컨테이너를 제거합니다. Docker 오류는 실패이며 메모리 DB로 대체하거나 건너뛰지 않습니다. 성공 결과는 `.proofstill/test-results/e2e.json`에 남고, 다음 E2E 시작 시 오래된 결과를 제거합니다. 브라우저 클릭은 이 API E2E의 범위에 포함하지 않습니다.

구성 근거: [Alembic 연결 공유](https://alembic.sqlalchemy.org/en/latest/cookbook.html#sharing-a-connection-across-one-or-more-programmatic-migration-commands), [Testcontainers PostgreSQL](https://testcontainers-python.readthedocs.io/en/latest/).

## 개발과 최종 리뷰

`init <기능 정의 JSON>` → 구현 및 `test <ID>` → `review <ID>` → `report <ID>` 순서로 진행합니다. test는 기본 Green, review는 기준 적절성과 구현을 함께 보는 최종 독립 리뷰입니다. 사전 criteria 리뷰와 Red 검사는 선택 사항입니다. 테스트 파일을 고치면 같은 작업에서 다시 검사하며 이전 결과는 현재 버전의 근거가 되지 않습니다.

리뷰는 원본 프로젝트를 읽고 대상 경로·해시만 보존합니다. 코드 복사본을 만들지 않으며, 리뷰 중 담당 소스나 테스트가 바뀌면 판정을 무효화합니다. 같은 작업 ID의 명령은 동시에 쓰지 못하게 잠그고, 서로 다른 작업은 담당 sources와 공유 의존성 범위로 검사합니다.

개발 중 `npm run test:integration` 또는 `npm run test:flow`, 통합 시 `npm run test:fast`를 사용합니다. flow 검사는 할 일 등록 → 완료 처리 → 재조회 CLI 흐름을 실행마다 별도 메모리 DB·임의 포트·임시 기록으로 검사합니다. 두 실행을 동시에 돌려 데이터가 섞이지 않는지도 확인합니다. Node/Python 의존성 설치는 필요하지만 Docker와 외부 DB는 필요하지 않습니다. 브라우저 클릭 검사는 포함하지 않습니다.

병렬 구현은 앱·기능별 폴더로 나눕니다. UI 기능은 `apps/app/src/features/<기능>`에 두고, 백엔드는 domain의 기능 파일 및 대응 infra·테스트 파일을 담당합니다. API 계약·생성 코드·공유 UI·migration·package/lockfile·진입점 연결은 한 담당자가 통합합니다. Git worktree는 이러한 경계와 실행별 자원 분리로 해결되지 않을 때만 고려합니다.

ORM 필드 추가 → `db:revision -- --message "change" --autogenerate` → migration 검토·적용 → 라우트 추가의 실행 예제는 [생성 프로젝트 안내](templates/monorepo/README.md#모델에서-api까지-확장하기)에 있습니다. OpenAPI는 수동 명세 대신 실제 FastAPI 앱에서 생성하며 export는 DB에 접속하지 않습니다. 구성 근거: [FastAPI lifespan](https://fastapi.tiangolo.com/advanced/events/), [SQLAlchemy declarative models](https://docs.sqlalchemy.org/en/20/orm/declarative_tables.html).

## JWT 인증 기반

백엔드 선택 시 `npm run backend:setup` 후 `npm run auth:setup`을 실행합니다.
루트 `.env.example`을 `.env`로 복사해 설정할 수 있으며, auth:setup은 비어 있는
`JWT_SECRET_KEY`에 무작위 키를 생성하고 기존 키·다른 설정은 보존합니다. 생성한
`.env`는 커밋하지 않습니다. `JWT_TTL_SECONDS` 기본값은 3600초입니다.
루트 `.env`는 npm/pnpm 백엔드 명령에서 읽으며, 기존 환경 변수가 우선합니다.

로그인 처리에서 자격 확인을 마친 뒤 토큰을 발급합니다. 사용자 데이터 구조는
고정하지 않으며 `data` 객체에 보존합니다. 공통 `iat`·`exp`는 발급기가 관리하고
검증은 HS256 서명과 만료를 필수 확인합니다.

```python
from fastapi import Depends
from infra.auth import TokenService, require_token

# 로그인 서비스에서 자격 확인을 마친 후 호출합니다.
token = TokenService.from_env().issue({'user_id': 'your-user-id'})
data = TokenService.from_env().verify(token)

# create_app 안에서 보호할 라우트에 적용합니다.
@app.get('/account')
def account(data=Depends(require_token)):
    return data
```

요청은 `Authorization: Bearer <token>`을 사용합니다. 잘못되거나 만료된 토큰은
401, 키 설정 누락은 503입니다. 이 기반에는 계정 저장·비밀번호 검사·로그인 API·
refresh/revoke 정책이 포함되지 않습니다. 기존 예제 API는 공개 상태이며 필요한
라우트에 의존성을 붙여 보호합니다. JWT 데이터는 암호화되지 않으므로 비밀번호나
비밀 키를 payload에 넣지 않습니다.

from pathlib import Path
from urllib.parse import quote
from .contract import HarnessError


def link(label, path):
    return f'[{label}]({quote(str(path), safe="/:?#=&%")})'


def render(h, plan, state, status):
    evidence = h.folder(plan['id'])
    current_source = h.source_digest(plan)
    try:
        current_criteria = h.criteria_digest(plan, state)
    except HarnessError:
        current_criteria = None
    criteria_valid = current_criteria is not None
    def current_run(run, phase='green'):
        return bool(criteria_valid and h.evidence_current(run) and run['criteria_digest'] == current_criteria
                    and (phase == 'red' or run['source_digest'] == current_source))
    lines = [f"# {plan['title']} — 기능 개발 보고", '',
             f"- 상태: **{'검증 완료' if status['complete'] else '미완료'}**", '- 사용자 수용: 미확인',
             f"- 의도: {plan['intent']}", f"- 업무 맥락: {plan['context']}",
             f"- 명령 실행 위치: `{h.root}`",
             f"- 기준: {link('고정된 요구사항·명령', evidence / 'plan.json')}",
             f"- 기준 버전: `{current_criteria}`", f"- 소스 버전: `{current_source}`", '',
             '## 요구사항과 검증', '', '| 요구사항 | 상태 | 테스트·근거 |', '|---|---|---|']
    for item in plan['requirements']:
        refs = []
        passed = True
        for check_id in item['checks']:
            run = h.latest(state, 'green', check_id)
            current = bool(current_run(run) and run['accepted'])
            passed = passed and current
            refs.append(link(check_id, run['log']) if run else f'{check_id}: 미실행')
        label = '자동 검사 통과' if passed else '실패/미검증'
        lines.append(f"| {item['id']}: {item['text'].replace('|', '/')} | {label} | {', '.join(refs)} |")
    lines += ['', '## 재현 명령과 검사 결과', '']
    for check in plan['checks']:
        lines += [f"### {check['id']} · {check['kind']} · {check['module']}", '',
                  '```sh', shlex.join(check['argv']), '```', '']
        for phase in ['red', 'green']:
            run = h.latest(state, phase, check['id'])
            if run:
                valid = current_run(run, phase)
                label = '현재 버전에서 무효' if not valid else ('기대 결과 확인' if run['accepted'] else '실패/미검증')
                lines.append(f"- {phase.title()}: {label} · 종료 코드 `{run['exit_code']}` · {link('출력', run['log'])} · {link('실행 영수증', run['receipt'])}")
            else:
                lines.append(f"- {phase.title()}: {'선택 검사 · 미실행' if phase == 'red' else '미실행'}")
        if check.get('trace_expectation'):
            lines.append('- ' + link('고정된 기대 흐름', h.root / check['trace_expectation']))
        run = h.latest(state, 'green', check['id'])
        for trace in (run or {}).get('traces', []):
            viewer = link('UI 추적 보기', trace['html']) if trace.get('html') else 'UI 뷰어 미생성'
            lines.append(f"- 흐름 {trace['name']} · {trace['steps']}단계 · {viewer} · {link('원본 JSONL', trace['path'])}")
            if 'state_machine' in trace:
                coverage = trace['state_machine']
                lines.append('- 상태머신 실행 전이: ' + ', '.join(coverage['visited_transitions']))
                lines.append('- 상태머신 미실행 전이: ' + (', '.join(coverage['unvisited_transitions']) or '없음'))
            for issue in trace['issues']:
                lines.append(f'  - 흐름 검증: {issue}')
        for path in check['files']:
            item = h.root / path
            if item.is_file():
                lines.append(f"- {link('테스트 코드', item)}")
            elif item.is_dir():
                for test in sorted(item.rglob('*')):
                    if test.is_file() and '__pycache__' not in test.parts:
                        lines.append(f"- {link('테스트 ' + test.name, test)}")
    lines += ['', '## 독립 리뷰', '']
    for stage, title in [('criteria', '사전 기준 검토 · 선택'), ('implementation', '기준·구현 최종 검토')]:
        record = state['reviews'].get(stage)
        if not record:
            lines.append(f'- {title}: 미실행')
            continue
        folder = Path(record['folder'])
        valid = criteria_valid and record['criteria_digest'] == current_criteria and (stage == 'criteria' or record['source_digest'] == current_source)
        if stage == 'implementation':
            valid = valid and all(current_run(h.latest(state, 'green', c['id']))
                                  and h.latest(state, 'green', c['id'])['accepted'] for c in plan['checks'])
        label = '현재 버전에서 무효' if not valid else ('통과' if record['passed'] else '실패/미검증')
        lines += [f"- {title}: {label} · 세션 `{record.get('session', '미확인')}` · 실행기 `{record['launcher']}`"]
        for name in ['answer.json', 'receipt.json', 'events.jsonl', 'input.txt', 'manifest.json']:
            path = folder / name
            if path.is_file():
                lines.append('  - ' + link(name, path))
        for check in record.get('answer', {}).get('checks', []):
            lines.append(f"  - {check.get('id')}: {check.get('status')} — {check.get('evidence')}")
    lines += ['', '## 구현·아티팩트', '']
    for artifact in plan['artifacts']:
        if artifact.get('url'):
            lines.append('- ' + link(artifact['label'], artifact['url']))
        else:
            path = (h.root / artifact['path']).resolve()
            lines.append('- ' + (link(artifact['label'], path) if path.is_file() else f"누락: {artifact['label']} — {path}"))
    lines += ['', '## 한계·남은 사항', '']
    limitations = list(plan['limitations']) + status['reasons']
    for record in state['reviews'].values():
        limitations += record.get('answer', {}).get('limitations', [])
    for limitation in dict.fromkeys(limitations):
        lines.append('- ' + limitation)
    lines.append('- 검증은 위 기준과 대상 버전에 한정하며, 모든 버그의 부재나 사용자의 최종 수용을 뜻하지 않는다.')
    from .handoff import snapshot, markdown
    overview = markdown(snapshot(h, plan['id']))
    lines[2:2] = [overview, '']
    return '\n'.join(lines) + '\n'
